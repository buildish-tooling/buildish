"page" -}} {{- $hasPipelinePage = true -}} {{- with index . "source" -}} {{- with index . "repository" }}{{ $repository = strings.TrimSuffix "/" . }}{{ end -}} {{- with index . "path" }}{{ $sourcePath = strings.TrimPrefix "/" . }}{{ end -}} {{- with index . "viewRef" }}{{ $viewRef = strings.Trim . "/" }}{{ end -}} {{- with index . "editRef" }}{{ $editRef = strings.Trim . "/" }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- $isStagedFile := false -}} {{- $nativePath := "" -}} {{- with $page.File -}} {{- $filename := replace .Filename "\\" "/" -}} {{- $workingDir := replace hugo.WorkingDir "\\" "/" -}} {{- $isStagedFile = or (strings.Contains $filename "/.stage/") (strings.HasPrefix $filename ".stage/") -}} {{- $nativePath = strings.TrimPrefix (add $workingDir "/") $filename -}} {{- $nativePath = strings.TrimPrefix "/" $nativePath -}} {{- end -}} {{- if and (not $hasPipelinePage) (not $isStagedFile) $nativePath -}} {{- with $page.Param "github_repo" }}{{ $repository = strings.TrimSuffix "/" . }}{{ end -}} {{- $nativeRef := $page.Param "github_branch" | default "main" -}} {{- $viewRef = strings.Trim $nativeRef "/" -}} {{- $editRef = $viewRef -}} {{- $nativeSubdir := $page.Param "github_subdir" | default "" -}} {{- $sourcePath = path.Join $nativeSubdir $nativePath -}} {{- end -}} {{- $supportedRepository := "" -}} {{- with $repository -}} {{- $parsedRepository := urls.Parse . -}} {{- $repositoryPathParts := split (strings.Trim $parsedRepository.Path "/") "/" -}} {{- if and (eq $parsedRepository.Scheme "https") (eq $parsedRepository.Host "github.com") (eq (len $repositoryPathParts) 2) (not $parsedRepository.RawQuery) (not $parsedRepository.Fragment) -}} {{- $supportedRepository = strings.TrimSuffix ".git" . -}} {{- end -}} {{- end -}} {{- $viewURL := "" -}} {{- if and $supportedRepository $sourcePath $viewRef -}} {{- $escapedSourcePath := partial "buildish-url-path-escape.html" $sourcePath -}} {{- $escapedViewRef := partial "buildish-url-path-escape.html" $viewRef -}} {{- $viewURL = printf "%s/blob/%s/%s" $supportedRepository $escapedViewRef $escapedSourcePath -}} {{- end -}} {{- $editURL := "" -}} {{- $newPageURL := "" -}} {{- if and $supportedRepository $sourcePath $editRef -}} {{- $escapedSourcePath := partial "buildish-url-path-escape.html" $sourcePath -}} {{- $escapedEditRef := partial "buildish-url-path-escape.html" $editRef -}} {{- $editURL = printf "%s/edit/%s/%s" $supportedRepository $escapedEditRef $escapedSourcePath -}} {{- with resources.Get "stubs/new-page-template.md" -}} {{- $newPageQuery := querify "value" .Content "filename" "change-me.md" -}} {{- $newPageURL = printf "%s/new/%s/%s?%s" $supportedRepository $escapedEditRef (path.Dir $escapedSourcePath) $newPageQuery -}} {{- end -}} {{- end -}} {{- return (dict "repository" $supportedRepository "sourcePath" $sourcePath "viewURL" $viewURL "editURL" $editURL "newPageURL" $newPageURL ) -}}
