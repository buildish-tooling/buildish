"slug" }}{{ $slug = . }}{{ end -}} {{- end -}} {{- if $slug -}} {{- $pagesBySlug = merge $pagesBySlug (dict $slug .) -}} {{- end -}} {{- end -}} {{- $catalogEntries := slice -}} {{- with index hugo.Data "components" -}} {{- with .items -}} {{- range $index, $item := . -}} {{- $slug := "" -}} {{- with index $item "slug" }}{{ $slug = . }}{{ end -}} {{- if $slug -}} {{- $weight := 0 -}} {{- with index $item "weight" }}{{ $weight = int . }}{{ end -}} {{- $catalogEntries = $catalogEntries | append (dict "slug" $slug "orderKey" (printf "%012d:%06d:%s" (add $weight 1000000) $index $slug) ) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $orderedPages := slice -}} {{- $seen := dict -}} {{- range sort $catalogEntries "orderKey" -}} {{- $slug := index . "orderKey" .RelPermalink ) -}} {{- end -}} {{- end -}} {{- range sort $fallbackEntries "orderKey" -}} {{- $orderedPages = $orderedPages | append (index . "page") -}} {{- end -}} {{- return $orderedPages -}}
